package org.bremersee.spring.security.ldaptive.userdetails;

import org.bremersee.spring.security.core.NormalizedPrincipalDetails;
import org.immutables.serial.Serial;
import org.immutables.value.Value;
import org.jspecify.annotations.NonNull;
import org.jspecify.annotations.NullMarked;
import org.jspecify.annotations.Nullable;

/**
 * The ldaptive user details.
 *
 * @author Christian Bremer
 */
@Value.Style(
    visibility = Value.Style.ImplementationVisibility.PACKAGE,
    overshadowImplementation = true,
    depluralize = true,
    jdk9Collections = true,
    get = {"get*", "is*"})
@Value.Immutable
@Serial.Version(1L)
public interface LdaptiveUserDetails extends NormalizedPrincipalDetails {

  /**
   * Gets the distinguished name.
   *
   * @return the distinguished name
   */
  String getDn();

  @Nullable
  @Override
  String getPassword();

  @Override
  @NullMarked
  String getUsername();

  @Nullable
  @Override
  String getFirstName();

  @Nullable
  @Override
  String getLastName();

  @Nullable
  @Override
  String getEmail();

  /**
   * Empty ldaptive user details.
   *
   * @return the ldaptive user details
   */
  @NonNull
  static LdaptiveUserDetails empty() {
    return builder()
        .username("")
        .dn("")
        .build();
  }

  /**
   * Gets builder.
   *
   * @return the builder
   */
  static Builder builder() {
    return new Builder();
  }

  /**
   * The immutable builder.
   */
  class Builder extends ImmutableLdaptiveUserDetails.Builder {

    /**
     * Instantiates a new builder.
     */
    public Builder() {
      super();
    }

  }

}
